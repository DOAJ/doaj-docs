218 @classmethod 

219 def flash(cls, tup): 

220 if isinstance(tup, tuple): 

221 flash(tup[0], tup[1]) 

222 else: 

223 flash(tup) 

224 

225 @classmethod 

226 def flash_with_url(cls, message, category): 

227 flash(message, category + '+contains-url') 

228 

229 @classmethod 

230 def flash_with_param(cls, message, category="error", **kwargs): 

231 for key, value in kwargs.items(): 

232 if key != "message" or key != "category": 

233 message = message + (" {0}: {1},".format(key, value)) 

234 flash(message, category)
